    for (var x3 = 0; x3 <= w; x3 += 3){
      var u3 = x3 / w;
      var y3 = h * 0.12 + (0.62 - Math.sin(u3 * 5.1 + t * 1.1) * 0.10 - Math.sin(u3 * 9.7 + t * 0.6) * 0.07) * h * 0.78;
      if (x3 === 0) ctx.moveTo(x3, y3); else ctx.lineTo(x3, y3);
    }
    ctx.strokeStyle = "rgba(" + tc.mint + ",.7)";
    ctx.lineWidth = 1.5;
    ctx.stroke();

    var endY = goldY(w);
    ctx.beginPath();
    ctx.arc(w - 3, endY, 3.6, 0, Math.PI * 2);
    ctx.fillStyle = "rgba(" + tc.gold + ",1)";
    ctx.shadowBlur = 12;
    ctx.shadowColor = "rgba(" + tc.gold + ",.9)";
    ctx.fill();
    ctx.shadowBlur = 0;
  }

  /* --- PULSE --- */
  var pulseBars = [];
  for (var pb = 0; pb < 22; pb++) {
    pulseBars.push({ h: Math.random(), target: 0.15 + Math.random() * 0.85 });
  }
  function roundRect(ctx, x, y, w, h, r){
    r = Math.min(r, w / 2, h / 2);
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }
  function drawPulse(){
    if (!pulseCanvas || !fitCanvas(pulseCanvas)) return;
    var w = pulseCanvas.clientWidth, h = pulseCanvas.clientHeight;
    var ctx = pctx;
    ctx.clearRect(0, 0, w, h);

    var cg = ctx.createRadialGradient(w/2, h/2, 0, w/2, h/2, Math.max(w, h) * 0.6);
    cg.addColorStop(0, "rgba(" + tc.gold + ",.14)");
    cg.addColorStop(1, "rgba(" + tc.gold + ",0)");
    ctx.fillStyle = cg;
    ctx.fillRect(0, 0, w, h);

    var n = pulseBars.length;
    var gap = 3;
    var bw = (w - (n - 1) * gap) / n;

    for (var i = 0; i < n; i++){
      var b = pulseBars[i];
      b.h += (b.target - b.h) * 0.14;
      if (Math.random() < 0.06) b.target = 0.12 + Math.random() * 0.88;

      var bh = Math.max(4, b.h * h * 0.78);
      var bx = i * (bw + gap);
      var by = (h - bh) / 2;

      var g2 = ctx.createLinearGradient(0, by, 0, by + bh);
      g2.addColorStop(0, "rgba(" + tc.mint + ",.95)");
      g2.addColorStop(1, "rgba(" + tc.gold + ",.95)");
      ctx.fillStyle = g2;
      roundRect(ctx, bx, by, bw, bh, Math.min(bw / 2, 3));
      ctx.fill();

      ctx.beginPath();
      ctx.arc(bx + bw / 2, by, 1.7, 0, Math.PI * 2);
      ctx.fillStyle = "rgba(" + tc.gold + ",.95)";
      ctx.fill();
    }
  }

  var lastT = performance.now(), elapsed = 0;
  function dashLoop(now){
    var dt = Math.min((now - lastT) / 1000, 0.05);
    lastT = now;
    elapsed += dt;
    if (currentView === "graph") drawGraph(elapsed);
    if (currentView === "pulse") drawPulse();
    requestAnimationFrame(dashLoop);
  }
  if (!reduceMotion) requestAnimationFrame(dashLoop);
  else drawGraph(0);

  /* --- DIGITAL --- */
  var digitalMain = document.getElementById("digitalMain");
  var digitalSub1 = document.getElementById("digitalSub1");
  var digitalSub2 = document.getElementById("digitalSub2");
  var digitalVal = 1284930;
  setInterval(function(){
    if (currentView !== "digital") return;
    digitalVal += Math.floor(Math.random() * 900) - 200;
    if (digitalVal < 900000) digitalVal = 900000 + Math.floor(Math.random() * 50000);
    if (digitalMain) digitalMain.textContent = digitalVal.toLocaleString("en-US");
    if (digitalSub1) digitalSub1.textContent = (Math.random() * 9 + 1).toFixed(2);
    if (digitalSub2) digitalSub2.textContent = (Math.random() * 60 + 30).toFixed(1) + "%";
  }, 380);

  /* --- ANALYTICS --- */
  var gaugeEls = document.querySelectorAll(".gauge");
  var chartBars = document.querySelectorAll("#dashChart .bar");
  var CIRC = 2 * Math.PI * 32;

  function setGauge(el, val) {
    val = Math.max(0, Math.min(100, val));
    var fill = el.querySelector(".g-fill");
    var valEl = el.querySelector(".g-val");
    if (fill) fill.style.strokeDashoffset = CIRC * (1 - val / 100);
    if (valEl) valEl.textContent = Math.round(val) + "%";
  }
  gaugeEls.forEach(function (el) {
    setGauge(el, parseFloat(el.dataset.gauge) || 80);
  });
  setInterval(function(){
    if (currentView !== "analytics") return;
    gaugeEls.forEach(function (el) {
      var base = parseFloat(el.dataset.gauge) || 80;
      setGauge(el, base + (Math.random() - 0.5) * 6);
    });
    chartBars.forEach(function (bar) {
      bar.style.setProperty("--h", (35 + Math.random() * 65).toFixed(0) + "%");
    });
  }, 1400);

  /* --- LIVE VISITORS --- */
  var visitorEl = document.getElementById("visitorCount");
  var visitors = 248;
  setInterval(function(){
    visitors += Math.round((Math.random() - 0.5) * 14);
    if (visitors < 120) visitors = 120 + Math.round(Math.random() * 20);
    if (visitors > 480) visitors = 480 - Math.round(Math.random() * 20);
    if (visitorEl) visitorEl.textContent = visitors;
  }, 1900);

  window.addEventListener("load", function(){
    if (graphCanvas) fitCanvas(graphCanvas);
  });

  /* ---------- SCROLL REVEAL ---------- */
  var revealEls = document.querySelectorAll("[data-reveal]");
  if ("IntersectionObserver" in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        if (!en.isIntersecting) return;
        var el = en.target;
        var sibs = Array.prototype.slice.call(el.parentNode.children);
        var idx = sibs.indexOf(el);
        el.style.transitionDelay = Math.min(idx, 6) * 70 + "ms";
        el.classList.add("in");
        io.unobserve(el);
      });
    }, { threshold: 0.1, rootMargin: "0px 0px -40px 0px" });
    revealEls.forEach(function (el) { io.observe(el); });
  } else {
    revealEls.forEach(function (el) { el.classList.add("in"); });
  }

  /* ---------- STEPS LINE DRAW ---------- */
  var stepsWrap = document.getElementById("stepsWrap");
  if (stepsWrap && "IntersectionObserver" in window) {
    var sio = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        if (en.isIntersecting) { stepsWrap.classList.add("in"); sio.unobserve(stepsWrap); }
      });
    }, { threshold: 0.2 });
    sio.observe(stepsWrap);
  }

  /* ---------- SPOTLIGHT ON FEATURE CARDS ---------- */
  if (!reduceMotion && window.matchMedia("(hover:hover)").matches) {
    document.querySelectorAll(".feat").forEach(function (f) {
      f.addEventListener("mousemove", function (e) {
        var r = f.getBoundingClientRect();
        f.style.setProperty("--mx", (e.clientX - r.left) + "px");
        f.style.setProperty("--my", (e.clientY - r.top) + "px");
      });
    });
  }

  /* ---------- COUNTERS ---------- */
  function runCounter(el) {
    var target = parseInt(el.dataset.count, 10) || 0;
    var suffix = el.dataset.suffix || "";
    var dur = 1500, t0 = null;
    function step(ts) {
      if (!t0) t0 = ts;
      var p = Math.min((ts - t0) / dur, 1);
      var eased = 1 - Math.pow(1 - p, 3);
      el.textContent = Math.round(target * eased) + (p === 1 ? suffix : "");
      if (p < 1) requestAnimationFrame(step);
    }
    requestAnimationFrame(step);
  }
  var counters = document.querySelectorAll("[data-count]");
  if ("IntersectionObserver" in window) {
    var cio = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        if (en.isIntersecting) { runCounter(en.target); cio.unobserve(en.target); }
      });
    }, { threshold: 0.6 });
    counters.forEach(function (c) { cio.observe(c); });
  } else { counters.forEach(runCounter); }

  /* ---------- YEAR ---------- */
  var y = document.getElementById("year");
  if (y) y.textContent = new Date().getFullYear();

})();
